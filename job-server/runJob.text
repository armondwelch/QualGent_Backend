/**
 * Check if a path is a directory
 */
function isDirectory(filePath) {
  try {
    return fs.statSync(filePath).isDirectory();
  } catch (error) {
    return false;
  }
}

/**
 * Parse iOS buildPath from appwright.config.ts - super simple approach
 */
function parseIosBuildPath() {
  const appwrightConfigPath = path.join(__dirname, 'appwright.config.ts');
  
  try {
    const configContent = fs.readFileSync(appwrightConfigPath, 'utf8');
    console.log(`[runJob.js] Reading config from: ${appwrightConfigPath}`);
    
    // Find all buildPath lines and look for the one with .ipa
    const lines = configContent.split('\n');
    
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i].trim();
      
      if (line.includes('buildPath:') && line.includes('.ipa')) {
        console.log(`[runJob.js] Found iOS buildPath at line ${i + 1}: ${line}`);
        
        // Extract the path.join part
        if (line.includes('path.join')) {
          const pathJoinMatch = line.match(/path\.join\(\s*["']([^"']+)["']\s*,\s*["']([^"']+)["']\s*\)/);
          if (pathJoinMatch) {
            const resolvedPath = path.join(pathJoinMatch[1], pathJoinMatch[2]);
            console.log(`[runJob.js] ✅ Parsed iOS buildPath: ${resolvedPath}`);
            return resolvedPath;
          }
        } else {
          // Handle simple string path
          const simplePathMatch = line.match(/["']([^"']+)["']/);
          if (simplePathMatch) {
            console.log(`[runJob.js] ✅ Parsed iOS buildPath: ${simplePathMatch[1]}`);
            return simplePathMatch[1];
          }
        }
      }
    }
    
    console.log('[runJob.js] ❌ Could not find iOS buildPath (looking for .ipa)');
    return null;
  } catch (error) {
    console.error(`[runJob.js] Error reading config: ${error.message}`);
    return null;
  }
}

/**
 * Upload app to BrowserStack manually and check for existing uploads
 */
async function uploadToBrowserStack(filePath) {
  const username = process.env.BROWSERSTACK_USERNAME;
  const accessKey = process.env.BROWSERSTACK_ACCESS_KEY;
  
  if (!username || !accessKey) {
    throw new Error('BrowserStack credentials not found');
  }

  // First, check if this app was already uploaded by checking recent uploads
  try {
    console.log(`[runJob.js] Checking for existing uploads...`);
    
    const stats = fs.statSync(filePath);
    const fileSizeMB = Math.round(stats.size / 1024 / 1024);
    const fileName = path.basename(filePath);
    
    // Get recent uploaded apps
    const existingApps = await getRecentBrowserStackApps();
    
    // Look for an app with same name and similar size (within 5MB)
    const existingApp = existingApps.find(app => {
      const appName = app.app_name || '';
      const appSize = app.app_size || 0;
      const appSizeMB = Math.round(appSize / 1024 / 1024);
      
      return appName.includes(fileName.replace('.ipa', '')) && 
             Math.abs(appSizeMB - fileSizeMB) <= 5;
    });
    
    if (existingApp) {
      console.log(`[runJob.js] ✅ Found existing upload: ${existingApp.app_url} (${Math.round(existingApp.app_size / 1024 / 1024)}MB)`);
      return existingApp.app_url;
    }
    
    console.log(`[runJob.js] No existing upload found, uploading ${fileName} (${fileSizeMB}MB)...`);
  } catch (checkError) {
    console.log(`[runJob.js] Could not check existing uploads: ${checkError.message}, proceeding with upload...`);
  }

  // Perform the upload
  return new Promise((resolve, reject) => {
    const FormData = require('form-data');
    const form = new FormData();
    form.append('file', fs.createReadStream(filePath));

    const options = {
      hostname: 'api-cloud.browserstack.com',
      path: '/app-automate/upload',
      method: 'POST',
      auth: `${username}:${accessKey}`,
      headers: form.getHeaders(),
      timeout: 300000 // 5 minute timeout for large files
    };

    console.log(`[runJob.js] Starting upload of ${path.basename(filePath)}...`);
    
    const req = https.request(options, (res) => {
      let data = '';
      
      res.on('data', (chunk) => {
        data += chunk;
      });
      
      res.on('end', () => {
        if (res.statusCode === 200) {
          try {
            const result = JSON.parse(data);
            console.log(`[runJob.js] ✅ Upload successful: ${result.app_url}`);
            return resolve(result.app_url);
          } catch (parseError) {
            return reject(new Error(`Failed to parse upload response: ${data}`));
          }
        } else {
          return reject(new Error(`Upload failed with status ${res.statusCode}: ${data}`));
        }
      });
    });

    req.on('error', (error) => {
      reject(new Error(`Upload request failed: ${error.message}`));
    });

    req.on('timeout', () => {
      req.destroy();
      reject(new Error('Upload timeout after 5 minutes'));
    });

    form.pipe(req);
  });
}

/**
 * Get recent uploaded apps from BrowserStack
 */
async function getRecentBrowserStackApps() {
  const username = process.env.BROWSERSTACK_USERNAME;
  const accessKey = process.env.BROWSERSTACK_ACCESS_KEY;
  
  if (!username || !accessKey) {
    return [];
  }

  return new Promise((resolve, reject) => {
    const auth = Buffer.from(`${username}:${accessKey}`).toString('base64');
    const options = {
      hostname: 'api-cloud.browserstack.com',
      path: '/app-automate/recent_apps',
      method: 'GET',
      headers: {
        'Authorization': `Basic ${auth}`,
        'User-Agent': 'NodeJS-Client'
      },
      timeout: 10000
    };

    const req = https.request(options, (res) => {
      let data = '';
      
      res.on('data', (chunk) => {
        data += chunk;
      });
      
      res.on('end', () => {
        if (res.statusCode === 200) {
          try {
            const result = JSON.parse(data);
            resolve(result || []);
          } catch (parseError) {
            console.log(`[runJob.js] Could not parse recent apps response`);
            resolve([]);
          }
        } else {
          console.log(`[runJob.js] Recent apps request failed: ${res.statusCode}`);
          resolve([]);
        }
      });
    });

    req.on('error', () => {
      resolve([]);
    });

    req.on('timeout', () => {
      req.destroy();
      resolve([]);
    });

    req.end();
  });
}

/**
 * Update config file to add app_url while keeping buildPath
 */
async function updateConfigWithAppUrl(appUrl) {
  const appwrightConfigPath = path.join(__dirname, 'appwright.config.ts');
  
  try {
    let configContent = fs.readFileSync(appwrightConfigPath, 'utf8');
    console.log(`[runJob.js] Updating config with app URL: ${appUrl}`);
    
    // More precise replacement - find the iOS buildPath line and add app URL after it
    const lines = configContent.split('\n');
    let updatedLines = [];
    let appUrlAdded = false;
    
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      
      // If this line contains buildPath and .ipa (iOS line), keep it and add app URL after
      if (line.includes('buildPath:') && line.includes('.ipa') && !appUrlAdded) {
        // Keep the original buildPath line
        updatedLines.push(line);
        
        // Extract the indentation from the original line
        const indentation = line.match(/^\s*/)[0];
        const appLine = `${indentation}app: "${appUrl}", // BrowserStack app URL`;
        console.log(`[runJob.js] Adding app URL after line ${i + 1}: ${appLine.trim()}`);
        updatedLines.push(appLine);
        appUrlAdded = true;
      } else if (line.includes('app:') && line.includes('bs://')) {
        // If there's already an app URL line, replace it
        const indentation = line.match(/^\s*/)[0];
        const newAppLine = `${indentation}app: "${appUrl}", // BrowserStack app URL`;
        console.log(`[runJob.js] Replacing existing app URL line ${i + 1}: ${newAppLine.trim()}`);
        updatedLines.push(newAppLine);
      } else {
        updatedLines.push(line);
      }
    }
    
    const finalConfig = updatedLines.join('\n');
    fs.writeFileSync(appwrightConfigPath, finalConfig, 'utf8');
    console.log(`[runJob.js] ✅ Config successfully updated (kept buildPath + added app URL)`);
    
  } catch (error) {
    console.error(`[runJob.js] Failed to update config: ${error.message}`);
    throw error;
  }
}

/**
 * Convert iOS .app to .ipa based on config and handle upload
 */
async function convertConfigDefinedApp(targetProject) {
  if (targetProject !== 'ios') return;

  const buildPath = parseIosBuildPath();
  if (!buildPath) {
    console.log('[runJob.js] Could not determine iOS buildPath');
    return;
  }

  console.log(`[runJob.js] � Target iOS buildPath: ${buildPath}`);

  // Check if .ipa already exists
  if (fs.existsSync(buildPath)) {
    const stats = fs.statSync(buildPath);
    console.log(`[runJob.js] ✅ iOS .ipa already exists: ${buildPath} (${Math.round(stats.size / 1024 / 1024)}MB)`);
    
    // Upload to BrowserStack and update config
    try {
      const appUrl = await uploadToBrowserStack(buildPath);
      await updateConfigWithAppUrl(appUrl);
      console.log(`[runJob.js] ✅ Config updated with BrowserStack app URL`);
    } catch (uploadError) {
      console.error(`[runJob.js] ❌ Upload failed: ${uploadError.message}`);
      // Continue anyway - maybe Appwright will work
    }
    
    return buildPath;
  }

  // Look for corresponding .app file
  const appPath = buildPath.replace(/\.ipa$/, '.app');
  
  if (fs.existsSync(appPath) && isDirectory(appPath)) {
    console.log(`[runJob.js] Converting ${appPath} to ${buildPath}`);
    
    try {
      const resultIpaPath = await convertAppToIpa(appPath);
      
      if (resultIpaPath !== buildPath) {
        fs.renameSync(resultIpaPath, buildPath);
      }
      
      const stats = fs.statSync(buildPath);
      console.log(`[runJob.js] ✅ Conversion successful: ${buildPath} (${Math.round(stats.size / 1024 / 1024)}MB)`);
      
      // Upload to BrowserStack and update config
      try {
        const appUrl = await uploadToBrowserStack(buildPath);
        await updateConfigWithAppUrl(appUrl);
        console.log(`[runJob.js] ✅ Config updated with BrowserStack app URL`);
      } catch (uploadError) {
        console.error(`[runJob.js] ❌ Upload failed: ${uploadError.message}`);
        // Continue anyway
      }
      
      return buildPath;
    } catch (error) {
      console.error(`[runJob.js] Conversion failed: ${error.message}`);
      throw error;
    }
  } else {
    console.log(`[runJob.js] ❌ Source .app not found: ${appPath}`);
    
    // Show available files for debugging
    const directory = path.dirname(buildPath);
    try {
      const files = fs.readdirSync(directory);
      console.log(`[runJob.js] Available files in ${directory}: ${files.join(', ')}`);
    } catch (e) {
      console.log(`[runJob.js] Could not read directory: ${directory}`);
    }
  }
}

/**
 * Get recent BrowserStack builds first, then get sessions from them
 */
async function getBrowserStackSessions() {
  const username = process.env.BROWSERSTACK_USERNAME;
  const accessKey = process.env.BROWSERSTACK_ACCESS_KEY;
  
  if (!username || !accessKey) {
    throw new Error('BrowserStack credentials not found');
  }
  
  console.log(`[runJob.js] Using BrowserStack username: ${username}`);
  
  if (httpClient) {
    // Use axios if available
    try {
      // First get builds
      const buildsResponse = await httpClient.get('https://api-cloud.browserstack.com/app-automate/builds.json?limit=5', {
        auth: {
          username: username,
          password: accessKey
        },
        headers: {
          'User-Agent': 'NodeJS-Client'
        },
        timeout: 10000
      });
      
      console.log(`[runJob.js] Builds response status: ${buildsResponse.status}`);
      
      // Get sessions from the most recent build
      if (buildsResponse.data && buildsResponse.data.length > 0) {
        const latestBuild = buildsResponse.data[0]; // Most recent build
        const buildId = latestBuild.automation_build.hashed_id;
        
        console.log(`[runJob.js] Getting sessions from build: ${buildId}`);
        
        // Now get sessions from this build
        const sessionsResponse = await httpClient.get(`https://api-cloud.browserstack.com/app-automate/builds/${buildId}/sessions.json`, {
          auth: {
            username: username,
            password: accessKey
          },
          headers: {
            'User-Agent': 'NodeJS-Client'
          },
          timeout: 10000
        });
        
        console.log(`[runJob.js] Sessions response status: ${sessionsResponse.status}`);
        return {
          sessions: sessionsResponse.data,
          buildId: buildId
        };
      } else {
        console.log('[runJob.js] No builds found');
        return { sessions: [], buildId: null };
      }
      
    } catch (error) {
      console.log(`[runJob.js] Axios error: ${error.message}`);
      console.log(`[runJob.js] Response status: ${error.response?.status}`);
      console.log(`[runJob.js] Response data: ${error.response?.data}`);
      throw error;
    }
  } else {
    // Simplified fallback
    return { sessions: [], buildId: null };
  }
}

/**
 * Test BrowserStack API connection
 */
async function testBrowserStackConnection() {
  return new Promise((resolve, reject) => {
    const username = process.env.BROWSERSTACK_USERNAME;
    const accessKey = process.env.BROWSERSTACK_ACCESS_KEY;
    
    if (!username || !accessKey) {
      reject(new Error('BrowserStack credentials not found'));

    const options = {

      method: 'GET',
      headers: {
        'Authorization': `Basic ${auth}`,
        'User-Agent': 'NodeJS-Client'
      }
    };
    
    const req = https.request(options, (res) => {
      let data = '';
      
      res.on('data', (chunk) => {
        data += chunk;
      });
      
      res.on('end', () => {
        if (res.statusCode === 200) {
          console.log('[runJob.js] BrowserStack API credentials are valid');
          resolve(data);
        } else {
          reject(new Error(`HTTP ${res.statusCode}: ${data}`));
        }
      });
    });
    
    req.on('error', reject);
    req.setTimeout(5000, () => {
      req.destroy();
      reject(new Error('Connection timeout'));
    });
    
    req.end();

/**
 * Runs an Appwright test job.
 * Executes the Appwright CLI command with the specified test path and config.
 * Automatically converts iOS .app bundles to .ipa files if needed based on config.
 * @param {object} job - The job object containing properties like test_path and target.
 * @param {string} job.test_path - The path to the test file to be executed.
 * @param {string} job.target - The target project name (e.g., 'android', 'ios').
 * @returns {Promise<string>} Resolves with stdout on success, rejects with Error on failure.
 */
async function runJob(job) {
  return new Promise(async (resolve, reject) => {
    const testPathEscaped = job.test_path;
    const targetProject = job.target;
    const appwrightConfigPath = path.join(__dirname, 'appwright.config.ts');
    
    // Convert iOS app if needed based on config
    if (targetProject === 'ios') {
      console.log('[runJob.js] Preprocessing iOS builds...');
      try {
        await convertConfigDefinedApp(targetProject);
      } catch (error) {
        console.error(`[runJob.js] iOS preprocessing failed: ${error.message}`);
        // Continue with test execution even if conversion fails
      }
    }
    
    const command = `node ${config.APPWRIGHT_PATH} test ${testPathEscaped} --config ${appwrightConfigPath} --project ${targetProject}`;
    
    console.log(`[runJob.js] Executing command: ${command}`);
    
    exec(command, { timeout: 60000 }, async (error, stdout, stderr) => {
      const fullOutput = stdout + stderr;
      const testsPassed = fullOutput.match(/^\s*\*?\s*✓/m) || fullOutput.includes('1 passed');
      
      if (error && !testsPassed) {
        console.error(`[runJob.js] Error running Appwright test: ${error.message}`);
        console.error(`[runJob.js] STDOUT: \n${stdout}`);
        console.error(`[runJob.js] STDERR: \n${stderr}`);
        
        if (fullOutput.includes("BROWSERSTACK_USERNAME") && fullOutput.includes("BROWSERSTACK_ACCESS_KEY")) {
          reject(new Error("Test failed: BrowserStack credentials required. This indicates your appwright.config.js might not be correctly loaded or there's an implicit BrowserStack dependency."));
        } else {
          reject(new Error(`Test failed during execution: ${error.message}`));
        }
      } else {
        console.log('[runJob.js] Test completed successfully.');
        
        // Try to get BrowserStack session details after test completion
        try {
          console.log('[runJob.js] Fetching BrowserStack session details...');
          // Add a small delay to ensure session is registered
          await new Promise(resolve => setTimeout(resolve, 3000));
          
          // Get recent sessions from the latest build
          const sessionResult = await getBrowserStackSessions();
          const { sessions, buildId } = sessionResult;
          
          if (sessions && sessions.length > 0) {
            // Get the most recent session (assuming it's the one we just ran)
            const latestSession = sessions[0].automation_session;
            
            console.log('[runJob.js] Latest BrowserStack session:');
            console.log(`- Session ID: ${latestSession.hashed_id}`);
            console.log(`- Name: ${latestSession.name}`);
            console.log(`- Status: ${latestSession.status}`);
            console.log(`- Browser: ${latestSession.browser} ${latestSession.browser_version}`);
            console.log(`- OS: ${latestSession.os} ${latestSession.os_version}`);
            console.log(`- Duration: ${latestSession.duration} seconds`);
            
            if (latestSession.public_url) {
              console.log(`- Public URL: ${latestSession.public_url}`);
            }
            if (latestSession.video_url) {
              console.log(`- Video URL: ${latestSession.video_url}`);
            }
            
            // Build the dashboard URL
            const dashboardUrl = `https://app-automate.browserstack.com/dashboard/v2/sessions/${latestSession.hashed_id}`;
            console.log(`- Dashboard URL: ${dashboardUrl}`);
            
            // Video storage handling
            if (latestSession.video_url && videoStorage) {
              console.log('[runJob.js] � Starting video download and storage...');
              
              try {
                const videoResult = await videoStorage.processBrowserStackSession(latestSession, buildId);
                
                if (videoResult.success) {
                  if (videoResult.existing) {
                    const sizeMB = Math.round(videoResult.size / 1024 / 1024);
                    console.log(`[runJob.js] ✅ Video already exists in database (${sizeMB}MB)`);
                  } else {
                    const sizeMB = Math.round(videoResult.size / 1024 / 1024);
                    console.log(`[runJob.js] ✅ Video data stored successfully!`);
                    console.log(`[runJob.js] � Database ID: ${videoResult.dbId}, Size: ${sizeMB}MB`);
                  }
                } else {
                  console.log(`[runJob.js] ❌ Video download failed: ${videoResult.error}`);
                }
              } catch (videoError) {
                console.log(`[runJob.js] ❌ Error processing video: ${videoError.message}`);
              }
            } else if (latestSession.video_url && !videoStorage) {
              console.log('[runJob.js] � Video URL available but video storage module not loaded');
            } else {
              console.log('[runJob.js] ⚠️  No video URL available for download');
            }
            
          } else {
            console.log('[runJob.js] No recent BrowserStack sessions found');
            
            // Try to get builds to see if there are any
            try {
              const builds = await getBrowserStackBuilds();
              console.log(`[runJob.js] Found ${builds.length} builds`);
              
              if (builds.length > 0) {
                console.log('[runJob.js] Recent builds:');
                builds.slice(0, 3).forEach((build, index) => {
                  console.log(`  ${index + 1}. ${build.automation_build.name} (${build.automation_build.hashed_id})`);
                });
              }
            } catch (buildsError) {
              console.log(`[runJob.js] Could not fetch builds: ${buildsError.message}`);
            }
          }
          
        } catch (sessionError) {
          console.log(`[runJob.js] Could not fetch BrowserStack session details: ${sessionError.message}`);
          
          // Try alternative approach - test credentials first
          try {
            console.log('[runJob.js] Testing BrowserStack API credentials...');
            await testBrowserStackConnection();
          } catch (credError) {
            console.log(`[runJob.js] Credential test failed: ${credError.message}`);
          }
        }
        
        resolve(stdout);
      }
    });
  });
}
